cmake_minimum_required(VERSION 3.20)
project(HonkaiCaptureNativeSmoke LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The pinned MediaPipe runtime is Windows x64 only")
endif()
add_executable(NativeSdkSmoke NativeSdkSmoke.cpp ../Source/HonkaiMotionCapture/Private/NativeMediaPipe.cpp)
target_compile_features(NativeSdkSmoke PRIVATE cxx_std_17)
target_compile_options(NativeSdkSmoke PRIVATE /EHsc /W4 /utf-8)
target_include_directories(NativeSdkSmoke PRIVATE ../Source/HonkaiMotionCapture/Private ../Source/ThirdParty/MediaPipe/include)
set(MEDIAPIPE_RUNTIME_DIR "" CACHE PATH "Prepared local directory containing libmediapipe.dll and Models")
set(FIXTURE_IMAGE "" CACHE FILEPATH "Local RGB image for SDK inference")
enable_testing()
if(MEDIAPIPE_RUNTIME_DIR AND FIXTURE_IMAGE)
    add_test(NAME NativeSdkInMemorySmoke COMMAND NativeSdkSmoke "${MEDIAPIPE_RUNTIME_DIR}" "${FIXTURE_IMAGE}")
endif()
